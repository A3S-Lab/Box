#[tokio::main]
async fn main() -> Result<(), a3s_box_sdk::ClientError> {

<Tab label="TypeScript" value="typescript">

```typescript
import { Sandbox } from '@a3s-lab/box';

async function main(): Promise<void> {
  const sandbox = await Sandbox.create('alpine:3.20');

  try {
    const result = await sandbox.commands.run('printf ready');
    console.log(result.stdout);
  } finally {
    await sandbox.kill();
  }
}

main().catch((error: unknown) => {
  console.error(error);
  process.exitCode = 1;
});
```

import asyncio

    async with await AsyncSandbox.create("alpine:3.20") as sandbox:

<Tab label="Go" value="go">

```go
package main

import (
    "context"
    "fmt"
    "log"

    box "github.com/A3S-Lab/Box/sdk/go/v3"
)

func main() {
    ctx := context.Background()
    sandbox, err := box.Create(ctx, "alpine:3.20")
    if err != nil {
        log.Fatal(err)
    }
    defer sandbox.Close(context.Background())

    result, err := sandbox.Run(ctx, box.Argv("printf", "ready"))
    if err != nil {
        log.Fatal(err)
    }

    fmt.Println(result.StdoutString())
}
```

import { Sandbox } from '@a3s-lab/box';

  const sandbox = await Sandbox.create('alpine:3.20');

  try {
    await sandbox.files.write('/workspace/status.txt', 'ready\n');
    console.log(await sandbox.files.read('/workspace/status.txt'));

    const result = await sandbox.commands.run('test -s /workspace/status.txt');

      throw new Error(result.stderr);
    }
